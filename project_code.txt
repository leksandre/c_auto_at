typedef struct {
  double x1;  // filter
  double x2;  // diff
  double r;   // track ratio
  double h0;  // filter coeff
  double h;   // update step
} TD_TypeDef;

class ADRC {
public:
    ADRC(){}

    void init_td_params(double r, double h, double td_n){
      td_.x1 = 0;
      td_.x2 = 0;
      td_.r = r;
      td_.h = h;
      td_.h0 = td_n*h;
    }

    void fhan_td(double x){
      double x1_delta = td_.x1 - x; //x1-v(k)
      double d = td_.r * td_.h0 * td_.h0; //d = rh^2 ,用h0代替h,解决最速跟踪微分器速度超调问题
      double a0 = td_.h0 * td_.x2; //a0 = h*x2
      double y = x1_delta + a0; //y = x1 + a0
      double a1 = sqrt(d * (d + 8*abs(y))); // a1=sqrt(d*(d+8*ABS(y))])
      double a2 = a0 + sign(y) * (a1 - d)/2.0; // a2 = a0 + sign(y)*(a1-d)/2
      double a = (a0 + y) * fsg(y , d) + a2 * (1 - fsg(y, d));
      double fh = -td_.r * (a/d)*fsg(a,d) - td_.r * sign(a) * (1 - fsg(a, d)); //fh = -r*(a/d)*sgn(a) - r*sgn(a)*(1-sgn(a))
      td_.x1 += td_.h * td_.x2; //x1 = x1' + h*x2'
      td_.x2 += td_.h * fh; //x2 = x2' + h*fh
    }

    double get_td_x1(){
      return td_.x1;
    }

    double get_td_x2(){
      return td_.x2;
    }

private:
  // sign function
  inline int sign(double in){
    int out = 0;
    if (in>1E-6) {
      out = 1;
    } else if (in<-1E-6) {
      out = -1;
    }
    return out;
  }

  inline int fsg(double x, double d) {
    int out = 0;
	  out = (sign(x + d) - sign(x - d)) / 2;
	  return out;
  }

  TD_TypeDef td_;

};

/**
 * @class GuidanceController
 * @brief Основной контроллер расчёта команд ускорения
 * 
 * Этот класс инкапсулирует всю логику наведения:
 * - Proportional Navigation (PN)
 * - PI контроль с feedforward
 * - Intercept алгоритм с ADRC
 * - Оценка скорости (Vc)
 * - Ограничение ускорений
 * 
 * @thread_safe Частично (требует внешней синхронизации для shared state)
 * 
 * @example
 * ```cpp
 * GuidanceConfig config;
 * config.pn_gain_N = 3.0;
 * config.max_lat_acc = 3.0;
 * 
 * GuidanceController controller(config, 0.02);  // dt = 20ms
 * 
 * GuidanceOutput output = controller.computeAcceleration(
 *   u_los, lambda, Vc, dt
 * );
 * ```
 */
class GuidanceController {
public:
  /**
   * @brief Конструктор
   * @param config Конфигурация из YAML
   * @param dt Шаг времени (сек), обычно 0.02 для 50Hz
   */
  explicit GuidanceController(const GuidanceConfig& config, double dt = 0.02);
  
  /**
   * @brief Деструктор
   */
  ~GuidanceController() = default;
  
  /**
   * @brief Расчёт команд ускорения
   * @param u_los Нормированный LOS вектор (nav frame)
   * @param lambda Угловая скорость LOS (nav frame)
   * @param pitch_rad Тангаж дрона (rad) для оценки Vc
   * @param yaw_rad Рыскание дрона (rad) для трансформации
   * @return GuidanceOutput с командой ускорения
   */
  GuidanceOutput computeAcceleration(
    const Eigen::Vector3d& u_los,
    const Eigen::Vector3d& lambda,
    double pitch_rad,
    double yaw_rad
  );
  
  /**
   * @brief Расчёт команд ускорения (альтернативный метод с Vc)
   * @param u_los Нормированный LOS вектор
   * @param lambda Угловая скорость LOS
   * @param Vc Оценка скорости (m/s)
   * @param yaw_rad Рыскание дрона (rad)
   * @return GuidanceOutput с командой ускорения
   */
  GuidanceOutput computeAccelerationWithVc(
    const Eigen::Vector3d& u_los,
    const Eigen::Vector3d& lambda,
    double Vc,
    double yaw_rad
  );
  
  /**
   * @brief Оценка скорости на основе тангажа
   * @param pitch_rad Тангаж дрона (rad)
   * @param max_tilt_rad Максимальный тангаж (rad)
   * @return Оценка скорости (m/s)
   */
  double estimateVcFromPitch(double pitch_rad, double max_tilt_rad);
  
  /**
   * @brief Оценка скорости на основе ускорения
   * @param acc_sp Команда ускорения (m/s²)
   * @return Оценка скорости (m/s)
   */
  double estimateVcFromAcc(const Eigen::Vector3d& acc_sp);
  
  /**
   * @brief Сброс внутренних состояний (интегралы, фильтры)
   */
  void reset();
  
  /**
   * @brief Обновление конфигурации на лету
   * @param config Новая конфигурация
   */
  void updateConfig(const GuidanceConfig& config);
  
  /**
   * @brief Получение последней команды ускорения
   * @return Команда ускорения (nav frame)
   */
  Eigen::Vector3d getLastAccelerationCommand() const;
  
  /**
   * @brief Получение отладочной информации
   * @return GuidanceDebug с отладочными данными
   */
  GuidanceDebug getDebugInfo() const;
  
  /**
   * @brief Включение/выключение режима PI контроля
   * @param enable true = включить PI
   */
  void setPIControlEnabled(bool enable);
  
  /**
   * @brief Включение/выключение режима top attack
   * @param enable true = включить top attack
   */
  void setTopAttackMode(bool enable);
  
  /**
   * @brief Включение/выключение parallel acceleration
   * @param enable true = включить parallel acc
   */
  void setParallelAccEnabled(bool enable);

private:
  GuidanceConfig config_;
  double dt_;
  mutable std::mutex mutex_;  // Защита состояния
  
  // === Состояния PI контроллера ===
  double los_err_int_y_{0.0};
  double los_err_int_z_{0.0};
  
  // === Состояния Intercept ===
  double intercept_x_int_{0.0};
  double intercept_y_int_{0.0};
  double intercept_los_err_x_prev_{0.0};
  double intercept_los_err_y_prev_{0.0};
  double intercept_x_d_flt_{0.0};
  double intercept_y_d_flt_{0.0};
  
  // === ADRC экземпляры ===
  ADRC adrc_intercept_ax_td_;
  ADRC adrc_intercept_ay_td_;
  bool adrc_initialized_{false};
  
  // === Состояния скорости ===
  double vc_pre_{0.0};

  double a_front_sp_{0.0};
  double ay_sp_{0.0};

  Eigen::Vector3d acc_frd_sp_{Eigen::Vector3d::Zero()};
  
  // === Отладка ===
  GuidanceDebug debug_info_;
  
  // === Константы ===
  static constexpr double CONSTANTS_ONE_G = 9.80665;
  static constexpr double EPSILON = 1e-6;

   */
  Eigen::Vector3d addParallelAcceleration_(

  Eigen::Vector3d los_vector{Eigen::Vector3d::Zero()};      ///< LOS вектор (nav frame)
  Eigen::Vector3d los_velocity{Eigen::Vector3d::Zero()};    ///< Скорость изменения LOS
  Eigen::Vector3d los_angular_velocity{Eigen::Vector3d::Zero()}; ///< Угловая скорость LOS (lambda)
  Eigen::Matrix<double, 6, 6> covariance{
    Eigen::Matrix<double, 6, 6>::Identity()};  ///< Ковариация состояния
  uint64_t timestamp_us{0};                     ///< Timestamp последней оценки
  bool is_valid{false};                         ///< Валидность оценки
  bool is_converged{false};                     ///< Сходимость фильтра
};

  double residual_norm{0.0};                              ///< Норма остатка

  bool gyro_compensation_applied{false};                  ///< Применена ли компенсация гироскопа

/**
 * @class LosEstimator
 * @brief Оценщик LOS вектора на основе Kalman Filter
 * 
 * Модель состояния (6D):
 * - Позиция LOS (единичный вектор в навигационной системе): [u_n, v_n, w_n]
 * - Скорость изменения LOS: [u_dot_n, v_dot_n, w_dot_n]
 * 
 * Модель измерения (3D):
 * - Измеренный LOS вектор из камеры: [u_meas, v_meas, w_meas]
 * 
 * @thread_safe Да (все публичные методы защищены mutex)
 * 
 * @example
 * ```cpp
 * LosEstimatorConfig config;
 * config.process_noise_Q = 0.01;
 * config.measurement_noise_R = 0.1;
 * config.use_gyro_compensation = true;
 * 
 * LosEstimator estimator(config);
 * 
 * // В цикле обновления
 * estimator.predict(dt, omega_b);
 * estimator.update(z_meas);
 * LosEstimate estimate = estimator.getEstimate();
 * ```
 */
class LosEstimator {
public:
  /**
   * @brief Конструктор
   * @param config Конфигурация Kalman Filter
   */
  explicit LosEstimator(const LosEstimatorConfig& config = LosEstimatorConfig());
  
  /**
   * @brief Деструктор
   */
  ~LosEstimator() = default;
  
  /**
   * @brief Предсказание состояния KF
   * @param dt Шаг времени (сек)
   * @param omega_b Угловая скорость тела (body frame, rad/s)
   * @param R_nb Матрица вращения nav → body
   */
  void predict(double dt, const Eigen::Vector3d& omega_b, const Eigen::Matrix3d& R_nb);
  
  /**
   * @brief Обновление состояния KF по измерению
   * @param z_meas Измеренный LOS вектор (nav frame, нормированный)
   */
  void update(const Eigen::Vector3d& z_meas);
  
  /**
   * @brief Получение текущей оценки LOS
   * @return LosEstimate с оценёнными значениями
   */
  LosEstimate getEstimate() const;
  
  /**
   * @brief Получение оценённого LOS вектора
   * @return LOS вектор (nav frame)
   */
  Eigen::Vector3d getLosVector() const;
  
  /**
   * @brief Получение оценённой угловой скорости LOS (lambda)
   * @return Угловая скорость LOS (rad/s)
   */
  Eigen::Vector3d getLosAngularVelocity() const;
  
  /**
   * @brief Получение ковариации состояния
   * @return Матрица ковариации 6x6
   */
  Eigen::Matrix<double, 6, 6> getCovariance() const;
  
  /**
   * @brief Проверка сходимости фильтра
   * @return true если фильтр сошёлся
   */
  bool isConverged() const;
  
  /**
   * @brief Проверка валидности оценки
   * @return true если оценка валидна
   */
  bool isValid() const;
  
  /**
   * @brief Сброс фильтра к начальным условиям
   */
  void reset();
  
  /**
   * @brief Обновление конфигурации на лету
   * @param config Новая конфигурация
   */
  void updateConfig(const LosEstimatorConfig& config);
  
  /**
   * @brief Получение отладочной информации
   * @return LosEstimatorDebug с отладочными данными
   */
  LosEstimatorDebug getDebugInfo() const;
  
  /**
   * @brief Включение/выключение компенсации гироскопа
   * @param enable true = включить компенсацию
   */
  void setGyroCompensationEnabled(bool enable);
  
  /**
   * @brief Включение/выключение адаптивного шума
   * @param enable true = включить адаптацию
   */
  void setAdaptiveNoiseEnabled(bool enable);

private:
  LosEstimatorConfig config_;
  mutable std::mutex mutex_;  // Защита состояния
  
  // === Состояние KF ===
  Eigen::VectorXd x_;  // Вектор состояния (6x1): [u, v, w, u_dot, v_dot, w_dot]
  Eigen::MatrixXd P_;  // Ковариация (6x6)
  Eigen::MatrixXd Q_;  // Шум процесса (6x6)
  Eigen::MatrixXd R_;  // Шум измерения (3x3)
  Eigen::MatrixXd F_;  // Перехода состояния (6x6)
  Eigen::MatrixXd H_;  // Измерений (3x6)
  
  // === Флаги ===
  bool initialized_{false};
  bool converged_{false};
  uint64_t update_count_{0};
  
  // === Отладка ===
  LosEstimatorDebug debug_info_;
  
  // === Константы ===
  static constexpr double EPSILON = 1e-6;
  static constexpr int STATE_DIM = 6;
  static constexpr int MEASUREMENT_DIM = 3;
  static constexpr int CONVERGENCE_THRESHOLD = 50;  // Обновлений для сходимости
  
  // === Внутренние методы ===
  
  /**
   * @brief Инициализация матриц KF
   */
  void initMatrices_();
  
  /**
   * @brief Адаптация шумов Q/R на основе остатка
   * @param residual Вектор остатка (innovation)
   */
  void adaptNoise_(const Eigen::VectorXd& residual);
  
  /**
   * @brief Матрица косого симметричного вектора
   * @param v Вектор 3D
   * @return Кососимметричная матрица 3x3
   */
  Eigen::Matrix3d skewSymmetric_(const Eigen::Vector3d& v);
  
  /**
   * @brief Нормализация LOS вектора в состоянии
   */
  void normalizeLosState_();
  
  /**
   * @brief Вычисление угловой скорости LOS (lambda = u × u_dot)
   * @return Угловая скорость LOS
   */
  Eigen::Vector3d computeLosLambda_() const;
  
  /**
   * @brief Проверка валидности вектора
   * @param v Вектор
   * @return true если вектор валиден
   */
  bool isValidVector_(const Eigen::VectorXd& v) const;
};

GuidanceController::GuidanceController(const GuidanceConfig& config, double dt)
  : config_(config)
  , dt_(dt)
{
  // Вычисление линейных пределов для оценки скорости
  vc_linear_max_ = std::sqrt(std::tan(config_.max_lat_acc / CONSTANTS_ONE_G));
  vc_linear_min_ = std::sqrt(std::tan(config_.min_vc / CONSTANTS_ONE_G));
  vc_linear_k_ = (config_.max_vc - config_.min_vc) / (vc_linear_max_ - vc_linear_min_);
  
  // Инициализация ADRC
  initADRC_();
  
  // Сброс состояний
  reset();
}

GuidanceOutput GuidanceController::computeAcceleration(
  const Eigen::Vector3d& u_los,
  const Eigen::Vector3d& lambda,
  double pitch_rad,
  double yaw_rad
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  GuidanceOutput output;
  
  // Проверка валидности входных данных
  if (u_los.norm() < EPSILON) {
    output.is_valid = false;
    return output;
  }
  
  // Оценка скорости на основе тангажа
  double Vc = estimateVcFromPitch(pitch_rad, config_.max_lat_acc / CONSTANTS_ONE_G);
  output.Vc_estimate = Vc;
  
  // Выбор алгоритма наведения
  Eigen::Vector3d acc_cmd;
  if (config_.enable_PI_control) {
    if (config_.top_attack_mode) {
      acc_cmd = computeInterceptPI_(u_los, dt_);
    } else {
      acc_cmd = computePnPI_(u_los, lambda, Vc, dt_);
    }
  } else {
    // Базовый PN: a = N * Vc * (lambda × u_los)
    acc_cmd = config_.pn_gain_N * Vc * lambda.cross(u_los);
  }
  
  // Добавление parallel acceleration (для увеличения Vc)
  if (config_.enable_acc_parallel) {
    acc_cmd = addParallelAcceleration_(acc_cmd, u_los, Vc, dt_);
  }
  
  // Ограничение ускорений
  acc_cmd = clampAcceleration_(acc_cmd);
  
  // Трансформация в навигационную систему
  output.acceleration_cmd = transformAccToNav_(acc_cmd, yaw_rad);
  output.acceleration_frd = acc_cmd;
  output.is_valid = true;
  
  // Сохранение для последующего использования
  last_acc_cmd_ = output.acceleration_cmd;
  
  return output;
}

GuidanceOutput GuidanceController::computeAccelerationWithVc(
  const Eigen::Vector3d& u_los,
  const Eigen::Vector3d& lambda,
  double Vc,
  double yaw_rad
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  GuidanceOutput output;
  
  if (u_los.norm() < EPSILON) {
    output.is_valid = false;
    return output;
  }
  
  output.Vc_estimate = Vc;
  
  Eigen::Vector3d acc_cmd;
  if (config_.enable_PI_control) {
    if (config_.top_attack_mode) {
      acc_cmd = computeInterceptPI_(u_los, dt_);
    } else {
      acc_cmd = computePnPI_(u_los, lambda, Vc, dt_);
    }
  } else {
    acc_cmd = config_.pn_gain_N * Vc * lambda.cross(u_los);
  }
  
  if (config_.enable_acc_parallel) {
    acc_cmd = addParallelAcceleration_(acc_cmd, u_los, Vc, dt_);
  }
  
  acc_cmd = clampAcceleration_(acc_cmd);
  output.acceleration_cmd = transformAccToNav_(acc_cmd, yaw_rad);
  output.acceleration_frd = acc_cmd;
  output.is_valid = true;
  
  last_acc_cmd_ = output.acceleration_cmd;
  
  return output;
}

double GuidanceController::estimateVcFromPitch(double pitch_rad, double max_tilt_rad) {
  // 🆕 Компенсация ЦТ
  double pitch_compensated = pitch_rad;
  if (config_.enable_cg_compensation) {
    double cg_trim_rad = config_.cg_trim_pitch_deg * M_PI / 180.0;
    pitch_compensated = pitch_rad - cg_trim_rad;
  }
  
  double vc = 0.0;
  
  // 🆕 Масштабирование для тяжёлого дрона
  double effective_max_tilt = max_tilt_rad * config_.accel_scale;
  double effective_min_tilt = (config_.min_vc / CONSTANTS_ONE_G) * config_.accel_scale;
  
  if (-pitch_compensated > effective_max_tilt) {
    vc = config_.max_vc * config_.accel_scale;  // 🆕 Применяем масштаб
  } else if (-pitch_compensated < effective_min_tilt) {
    vc = config_.min_vc * config_.accel_scale;
  } else {
    double forward_acc = std::sqrt(std::tan(-pitch_compensated));
    vc = (forward_acc - vc_linear_min_) * vc_linear_k_ + config_.min_vc;
    vc *= config_.accel_scale;  // 🆕 Применяем масштаб
  }
  
  // 🆕 Больше сглаживание для тяжёлого
  vc = vc_pre_ + (config_.vc_alpha * config_.mass_ratio) * (vc - vc_pre_);
  vc_pre_ = vc;
  
  debug_info_.vc_estimate = vc;
  return vc;
}


double GuidanceController::estimateVcFromAcc(const Eigen::Vector3d& acc_sp) {
  // v = K * sqrt(acc)
  double acc = acc_sp.norm();
  double vc = 0.0;
  
  if (acc > EPSILON) {
    double x = std::sqrt(acc);
    vc = (x - vc_linear_min_) * vc_linear_k_ + config_.min_vc;
  } else {
    vc = config_.min_vc;
  }
  
  vc = vc_pre_ + config_.vc_alpha * (vc - vc_pre_);
  vc_pre_ = vc;
  
  debug_info_.vc_estimate = vc;
  
  return vc;
}

Eigen::Vector3d GuidanceController::computePnPI_(
  const Eigen::Vector3d& u_los,
  const Eigen::Vector3d& lambda,
  double Vc,
  double dt
) {
  // Ошибка LOS (lambda × u_los)
  Eigen::Vector3d los_err = lambda.cross(u_los);
  double los_err_y = los_err.y();
  double los_err_z = los_err.z();
  
  // Интеграл с anti-windup (Y axis)
  if (!((los_err_int_y_ >= config_.los_y_int_limit && los_err_y > 0.0) ||
        (los_err_int_y_ <= -config_.los_y_int_limit && los_err_y < 0.0))) {
    los_err_int_y_ += config_.los_y_ki * los_err_y * dt;
  }
  
  // Интеграл с anti-windup (Z axis)
  if (!((los_err_int_z_ >= config_.los_z_int_limit && los_err_z > 0.0) ||
        (los_err_int_z_ <= -config_.los_z_int_limit && los_err_z < 0.0))) {
    los_err_int_z_ += config_.los_z_ki * los_err_z * dt;
  }
  
  // Feedforward (Y axis) ✅ ИСПРАВЛЕНО
  double ay_pn_ff = 0.0;
  if (config_.los_y_ff_enable) {  // ← было los_y_ffenable
    if (u_los.y() > config_.los_y_err_ff_limit) {
      ay_pn_ff = config_.los_y_ff * (u_los.y() - config_.los_y_err_ff_limit);
    } else if (u_los.y() < -config_.los_y_err_ff_limit) {
      ay_pn_ff = config_.los_y_ff * (u_los.y() + config_.los_y_err_ff_limit);
    }
    ay_pn_ff = clamp_(ay_pn_ff, -config_.los_y_ff_limit, config_.los_y_ff_limit);  // ← было los_y_fflimit
  }
  
  // Feedforward (Z axis) ✅ ИСПРАВЛЕНО
  double az_pn_ff = 0.0;
  if (config_.los_z_ff_enable) {
    if (u_los.z() > config_.los_z_err_ff_limit) {
      az_pn_ff = config_.los_z_ff * (u_los.z() - config_.los_z_err_ff_limit);
    } else if (u_los.z() < -config_.los_z_err_ff_limit) {
      az_pn_ff = config_.los_z_ff * (u_los.z() + config_.los_z_err_ff_limit);
    }
    az_pn_ff = clamp_(az_pn_ff, -config_.los_z_ff_limit, config_.los_z_ff_limit);  // ← было los_z_fflimit
  }
  
  // PI контроль
  double a_pn_y = ay_pn_ff + config_.los_y_kp * Vc * los_err_y + los_err_int_y_;
  double a_pn_z = az_pn_ff + config_.los_z_kp * Vc * los_err_z + los_err_int_z_;
  
  // Ограничение
  a_pn_y = clamp_(a_pn_y, -config_.max_lat_acc, config_.max_lat_acc);
  a_pn_z = clamp_(a_pn_z, config_.max_up_acc, config_.max_down_acc);
  
  // Сохранение отладки
  debug_info_.los_y_ff = ay_pn_ff;
  debug_info_.los_z_ff = az_pn_ff;
  debug_info_.los_y_kp = config_.los_y_kp * Vc * los_err_y;
  debug_info_.los_z_kp = config_.los_z_kp * Vc * los_err_z;
  debug_info_.los_y_int = los_err_int_y_;
  debug_info_.los_z_int = los_err_int_z_;
  debug_info_.los_err_y = los_err_y;
  debug_info_.los_err_z = los_err_z;
  
  Eigen::Vector3d a_pn{0.0, a_pn_y, a_pn_z};
  return a_pn;
}






Eigen::Vector3d GuidanceController::computeInterceptPI_(
  const Eigen::Vector3d& u_los,
  const Eigen::Vector3d& lambda,
  double Vc,
  double dt
) {
  // Ошибка LOS
  Eigen::Vector3d los_err = lambda.cross(u_los);
  double los_err_x = los_err.x();
  double los_err_y = los_err.y();
  
  // Интеграл с anti-windup (X axis)
  if (!((intercept_x_int_ >= config_.intercept_int_limit && los_err_x > 0.0) ||
        (intercept_x_int_ <= -config_.intercept_int_limit && los_err_x < 0.0))) {
    intercept_x_int_ += config_.intercept_ki * los_err_x * dt;
  }
  
  // Интеграл с anti-windup (Y axis)
  if (!((intercept_y_int_ >= config_.intercept_int_limit && los_err_y > 0.0) ||
        (intercept_y_int_ <= -config_.intercept_int_limit && los_err_y < 0.0))) {
    intercept_y_int_ += config_.intercept_ki * los_err_y * dt;
  }
  
  // PI контроль
  double a_pn_x = config_.intercept_kp * Vc * los_err_x + intercept_x_int_;
  double a_pn_y = config_.intercept_kp * Vc * los_err_y + intercept_y_int_;
  double a_pn_z = config_.intercept_z_kp * u_los.z();
  
  Eigen::Vector3d a_pn{a_pn_x, a_pn_y, a_pn_z};
  return a_pn;
}

Eigen::Vector3d GuidanceController::computeInterceptPI_(
  const Eigen::Vector3d& u_los,
  double dt
) {
  // Ошибка LOS (без lambda)
  double los_err_x = u_los.x();
  double los_err_y = u_los.y();
  
  // Интеграл с anti-windup
  if (!((intercept_x_int_ >= config_.intercept_int_limit && los_err_x > 0.0) ||
        (intercept_x_int_ <= -config_.intercept_int_limit && los_err_x < 0.0))) {
    intercept_x_int_ += config_.intercept_ki * los_err_x * dt;
  }
  
  if (!((intercept_y_int_ >= config_.intercept_int_limit && los_err_y > 0.0) ||
        (intercept_y_int_ <= -config_.intercept_int_limit && los_err_y < 0.0))) {
    intercept_y_int_ += config_.intercept_ki * los_err_y * dt;
  }
  
  // Производная
  double intercept_x_d = (los_err_x - intercept_los_err_x_prev_) / dt;
  double intercept_y_d = (los_err_y - intercept_los_err_y_prev_) / dt;
  intercept_los_err_x_prev_ = los_err_x;
  intercept_los_err_y_prev_ = los_err_y;
  
  // Фильтрация производной
  intercept_x_d_flt_ = (1.0 - config_.intercept_acc_alpha) * intercept_x_d_flt_ +
                       config_.intercept_acc_alpha * intercept_x_d;
  intercept_y_d_flt_ = (1.0 - config_.intercept_acc_alpha) * intercept_y_d_flt_ +
                       config_.intercept_acc_alpha * intercept_y_d;
  
  // PID контроль
  double pn_vx = config_.intercept_kp * los_err_x + intercept_x_int_ +
                 config_.intercept_kd * intercept_x_d_flt_;
  double pn_vy = config_.intercept_kp * los_err_y + intercept_y_int_ +
                 config_.intercept_kd * intercept_y_d_flt_;
  double a_pn_z = config_.intercept_z_kp * u_los.z();
  
  // ADRC для сглаживания
  double a_pn_x = 0.0;
  double a_pn_y = 0.0;
  
  if (!adrc_initialized_) {
    initADRC_();
  } else {
    adrc_intercept_ax_td_.fhan_td(pn_vx);
    adrc_intercept_ay_td_.fhan_td(pn_vy);
    a_pn_x = adrc_intercept_ax_td_.get_td_x2();
    a_pn_y = adrc_intercept_ay_td_.get_td_x2();
  }
  
  // Сохранение отладки
  acc_frd_sp_ = {a_pn_x, a_pn_y, a_pn_z};
  
  Eigen::Vector3d a_pn{a_pn_x, a_pn_y, a_pn_z};
  return a_pn;
}

Eigen::Vector3d GuidanceController::clampAcceleration_(const Eigen::Vector3d& acc) {
  Eigen::Vector3d acc_clamped;
  
  // 🆕 Динамическое ограничение с учётом массы
  double mass_factor = config_.accel_scale;  // = 0.62 для 2519г
  
  acc_clamped.x() = clamp_(
    acc.x(),
    -config_.max_hor_acc * mass_factor,
    config_.max_hor_acc * mass_factor
  );
  acc_clamped.y() = clamp_(
    acc.y(),
    -config_.max_hor_acc * mass_factor,
    config_.max_hor_acc * mass_factor
  );
  acc_clamped.z() = clamp_(
    acc.z(),
    config_.max_up_acc * mass_factor,
    config_.max_down_acc * mass_factor
  );
  
  return acc_clamped;
}


Eigen::Vector3d GuidanceController::addParallelAcceleration_(
  const Eigen::Vector3d& acc,
  const Eigen::Vector3d& u_los,
  double Vc,
  double dt
) {
    (void)dt; 
  // Parallel acceleration для увеличения Vc
  double acc_parallel = config_.acc_parallel_gain * (config_.max_vc - Vc);
  acc_parallel = acc_parallel_pre_ + config_.acc_parallel_alpha * (acc_parallel - acc_parallel_pre_);
  acc_parallel = clamp_(acc_parallel, 0.0, config_.max_acc_parallel);
  acc_parallel_pre_ = acc_parallel;
  
  debug_info_.acc_parallel = acc_parallel;
  
  return acc + acc_parallel * u_los;
}

Eigen::Vector3d GuidanceController::transformAccToNav_(
  const Eigen::Vector3d& acc_frd,
  double yaw_rad
) {
  // Трансформация из body (FRD) в nav frame через yaw
  Eigen::Vector3d euler_local{0.0, 0.0, yaw_rad};
  Eigen::Quaterniond q_local = Eigen::AngleAxisd(euler_local.z(), Eigen::Vector3d::UnitZ()) *
                                Eigen::AngleAxisd(euler_local.y(), Eigen::Vector3d::UnitY()) *
                                Eigen::AngleAxisd(euler_local.x(), Eigen::Vector3d::UnitX());
  Eigen::Affine3d transformation(q_local);
  return transformation * acc_frd;
}

void GuidanceController::initADRC_() {
  adrc_intercept_ax_td_.init_td_params(config_.adrc_r, config_.adrc_h, config_.adrc_n);
  adrc_intercept_ay_td_.init_td_params(config_.adrc_r, config_.adrc_h, config_.adrc_n);
  adrc_initialized_ = true;
}

Eigen::Matrix3d GuidanceController::skewSymmetric_(const Eigen::Vector3d& v) {
  Eigen::Matrix3d m;
  m <<     0, -v.z(),  v.y(),
         v.z(),     0, -v.x(),
        -v.y(),  v.x(),     0;
  return m;
}

double GuidanceController::clamp_(double value, double min_val, double max_val) {
  return std::max(min_val, std::min(max_val, value));
}

LosEstimator::LosEstimator(const LosEstimatorConfig& config)
  : config_(config)
{
  // Инициализация матриц
  x_.setZero(STATE_DIM);
  P_.setIdentity(STATE_DIM, STATE_DIM);
  F_.setIdentity(STATE_DIM, STATE_DIM);
  H_.setZero(MEASUREMENT_DIM, STATE_DIM);
  
  initMatrices_();
  
  // Начальная ковариация
  P_.block<3, 3>(0, 0) *= config_.init_position_variance;
  P_.block<3, 3>(3, 3) *= config_.init_velocity_variance;
  
  initialized_ = true;
}

void LosEstimator::predict(
  double dt,
  const Eigen::Vector3d& omega_b,
  const Eigen::Matrix3d& R_nb
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (!initialized_) {
    return;
  }
  
  if (dt <= 0.0) {
    return;
  }
  
  debug_info_.gyro_compensation_applied = false;
  
  if (config_.use_gyro_compensation) {
    // === С компенсацией гироскопа ===
    // Конвертация угловой скорости из body frame в nav frame
    Eigen::Vector3d omega_n = R_nb.transpose() * omega_b;
    
    // Построение кососимметричной матрицы
    Eigen::Matrix3d omega_sym = skewSymmetric_(omega_n);
    
    // Построение матрицы перехода F с учётом вращения
    // Модель: u_dot = u_dot - ω × u
    F_.setIdentity();
    F_.block<3, 3>(0, 0) = Eigen::Matrix3d::Identity() - dt * omega_sym;
    F_.block<3, 3>(0, 3) = Eigen::Matrix3d::Identity() * dt;
    F_.block<3, 3>(3, 0) = -omega_sym;
    F_.block<3, 3>(3, 3) = Eigen::Matrix3d::Identity();
    
    // Предсказание состояния
    Eigen::Vector3d u = x_.segment<3>(0);
    Eigen::Vector3d u_dot = x_.segment<3>(3);
    
    // Компенсированная скорость: u_dot_comp = u_dot - ω × u
    Eigen::Vector3d u_dot_comp = u_dot - omega_n.cross(u);
    
    // Следующее состояние: u_next = u + dt * u_dot_comp
    Eigen::Vector3d u_next = u + dt * u_dot_comp;
    Eigen::Vector3d u_dot_next = u_dot_comp;
    
    // Нормализация LOS вектора
    double norm = u_next.norm();
    if (norm > EPSILON) {
      u_next.normalize();
    }
    
    // Обновление состояния
    x_.segment<3>(0) = u_next;
    x_.segment<3>(3) = u_dot_next;
    
    debug_info_.gyro_compensation_applied = true;
    
  } else {
    // === Без компенсации гироскопа (простая модель постоянной скорости) ===
    F_.setIdentity();
    F_.block<3, 3>(0, 3) = Eigen::Matrix3d::Identity() * dt;  // Позиция обновляется скоростью
    
    // Предсказание состояния: x = F * x
    x_ = F_ * x_;
    
    // Нормализация LOS вектора
    normalizeLosState_();
  }
  
  // Предсказание ковариации: P = F * P * F^T + Q
  P_ = F_ * P_ * F_.transpose() + Q_;
  
  // Обеспечение симметричности и положительной определённости P
  P_ = (P_ + P_.transpose()) * 0.5;
}

void LosEstimator::update(const Eigen::Vector3d& z_meas) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (!initialized_) {
    return;
  }
  
  // Проверка валидности измерения
  if (!isValidVector_(z_meas)) {
    return;
  }
  
  // Нормализация измерения
  Eigen::Vector3d z_normalized = z_meas.normalized();
  
  // Сохранение измерения для отладки
  debug_info_.measurement = z_normalized;
  
  // Вычисление остатка (innovation): y = z - H * x
  Eigen::VectorXd y = z_normalized - H_ * x_;
  debug_info_.residual = y;
  debug_info_.residual_norm = y.norm();
  
  // Вычисление ковариации остатка: S = H * P * H^T + R
  Eigen::MatrixXd S = H_ * P_ * H_.transpose() + R_;
  
  // Вычисление коэффициента Калмана: K = P * H^T * S^-1
  Eigen::MatrixXd K = P_ * H_.transpose() * S.inverse();
  
  // Сохранение диагонали K для отладки
  debug_info_.kalman_gain << K(0, 0), K(1, 1), K(2, 2);
  
  // Обновление состояния: x = x + K * y
  x_ = x_ + K * y;
  
  // Обновление ковариации: P = (I - K * H) * P
  Eigen::MatrixXd I = Eigen::MatrixXd::Identity(STATE_DIM, STATE_DIM);
  P_ = (I - K * H_) * P_;
  
  // Обеспечение симметричности P
  P_ = (P_ + P_.transpose()) * 0.5;
  
  // Нормализация LOS вектора в состоянии
  normalizeLosState_();
  
  // Адаптация шумов (если включена)
  if (config_.enable_adaptive_noise) {
    adaptNoise_(y);
  }
  
  // Обновление статистики
  update_count_++;
  if (update_count_ > CONVERGENCE_THRESHOLD) {
    converged_ = true;
  }
  
  // Сохранение отладочной информации
  debug_info_.trace_covariance = P_.trace();
}

LosEstimate LosEstimator::getEstimate() const {
  std::lock_guard<std::mutex> lock(mutex_);
  
  LosEstimate estimate;
  
  if (!initialized_) {
    estimate.is_valid = false;
    return estimate;
  }
  
  estimate.los_vector = x_.segment<3>(0).normalized();
  estimate.los_velocity = x_.segment<3>(3);
  estimate.los_angular_velocity = computeLosLambda_();
  estimate.covariance = P_;
  estimate.is_valid = true;
  estimate.is_converged = converged_;
  
  return estimate;
}

Eigen::Vector3d LosEstimator::getLosVector() const {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (!initialized_) {
    return Eigen::Vector3d::Zero();
  }
  
  return x_.segment<3>(0).normalized();
}

Eigen::Vector3d LosEstimator::getLosAngularVelocity() const {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (!initialized_) {
    return Eigen::Vector3d::Zero();
  }
  
  return computeLosLambda_();
}

Eigen::Matrix<double, 6, 6> LosEstimator::getCovariance() const {
  std::lock_guard<std::mutex> lock(mutex_);
  return P_;
}

bool LosEstimator::isConverged() const {
  std::lock_guard<std::mutex> lock(mutex_);
  return converged_;
}

bool LosEstimator::isValid() const {
  std::lock_guard<std::mutex> lock(mutex_);
  return initialized_ && isValidVector_(x_);
}

void LosEstimator::reset() {
  std::lock_guard<std::mutex> lock(mutex_);
  
  x_.setZero(STATE_DIM);
  P_.setIdentity(STATE_DIM, STATE_DIM);
  P_.block<3, 3>(0, 0) *= config_.init_position_variance;
  P_.block<3, 3>(3, 3) *= config_.init_velocity_variance;
  
  F_.setIdentity();
  H_.setZero(MEASUREMENT_DIM, STATE_DIM);
  H_.block<3, 3>(0, 0) = Eigen::Matrix3d::Identity();
  
  initialized_ = true;
  converged_ = false;
  update_count_ = 0;
  
  debug_info_ = LosEstimatorDebug();
}

  config_ = config;
  initMatrices_();
}

void LosEstimator::initMatrices_() {
  // Матрица измерения H: измеряем только позицию LOS (первые 3 состояния)
  H_.setZero(MEASUREMENT_DIM, STATE_DIM);
  H_.block<3, 3>(0, 0) = Eigen::Matrix3d::Identity();
  
  // Матрица шума процесса Q (диагональная)
  Q_ = config_.process_noise_Q * Eigen::MatrixXd::Identity(STATE_DIM, STATE_DIM);
  
  // Матрица шума измерения R (диагональная)
  R_ = config_.measurement_noise_R * Eigen::MatrixXd::Identity(MEASUREMENT_DIM, MEASUREMENT_DIM);
  
  // Матрица перехода F (инициализация, будет обновляться в predict)
  F_.setIdentity(STATE_DIM, STATE_DIM);
  F_.block<3, 3>(0, 3) = Eigen::Matrix3d::Identity();  // Constant velocity model
}

void LosEstimator::adaptNoise_(const Eigen::VectorXd& residual) {
  double res_norm = residual.norm();
  
  // Адаптивный масштаб: больший остаток → больше доверия к модели
  double scale = std::clamp(
    1.0 + res_norm,
    config_.adaptive_min_scale,
    config_.adaptive_max_scale
  );
  
  // Увеличиваем Q (меньше доверия к модели при большом остатке)
  Q_ *= scale;
  
  // Уменьшаем R (больше доверия к измерению при большом остатке)
  R_ /= scale;
  
  // Ограничение Q и R для стабильности
  Q_ = Q_.cwiseMax(EPSILON * Eigen::MatrixXd::Identity(STATE_DIM, STATE_DIM));
  R_ = R_.cwiseMax(EPSILON * Eigen::MatrixXd::Identity(MEASUREMENT_DIM, MEASUREMENT_DIM));
}

Eigen::Matrix3d LosEstimator::skewSymmetric_(const Eigen::Vector3d& v) {
  Eigen::Matrix3d m;
  m <<     0, -v.z(),  v.y(),
         v.z(),     0, -v.x(),
        -v.y(),  v.x(),     0;
  return m;
}

void LosEstimator::normalizeLosState_() {
  Eigen::Vector3d u = x_.segment<3>(0);
  double norm = u.norm();
  
  if (norm > EPSILON) {
    u.normalize();
    x_.segment<3>(0) = u;
  } else {
    // Если вектор слишком мал, инициализируем единичным
    x_.segment<3>(0) << 0.0, 0.0, 1.0;
  }
}

Eigen::Vector3d LosEstimator::computeLosLambda_() const {
  // lambda = u × u_dot (угловая скорость LOS)
  Eigen::Vector3d u = x_.segment<3>(0).normalized();
  Eigen::Vector3d u_dot = x_.segment<3>(3);
  return u.cross(u_dot);
}

bool LosEstimator::isValidVector_(const Eigen::VectorXd& v) const {
  if (v.size() < MEASUREMENT_DIM) {
    return false;
  }
  
  for (int i = 0; i < v.size(); ++i) {
    if (std::isnan(v(i)) || std::isinf(v(i))) {
      return false;
    }
  }
  
  return true;
}

/**
 * @brief Главная функция ROS2 узла
 * 
 * @threading_model MultiThreadedExecutor
 * - Callbacks обрабатываются в параллельных потоках
 * - Timer callback выполняется в отдельном потоке
 * - Shared state защищён mutex в контроллерах
 * 
 * @param argc Количество аргументов командной строки
 * @param argv Аргументы командной строки
 * @return int Код возврата (0 = успех)
 */


int main(int argc, char **argv) {
  // === 1. Инициализация ROS2 ===
  rclcpp::init(argc, argv);
  
  // === 2. Создание узла ===
  // Используем shared_ptr для совместимости с Executor
  auto los_node = std::make_shared<los_guidance::Node>();
  
  los_node->postConstruct();

  // === 3. Логирование старта ===
  RCLCPP_INFO(los_node->get_logger(), "===========================================");
    
  RCLCPP_INFO(los_node->get_logger(), "Threading Model: MultiThreadedExecutor");
  RCLCPP_INFO(los_node->get_logger(), "===========================================");
  
  // === 4. Создание MultiThreadedExecutor ===
  // Преимущества перед rclcpp::spin():
  // - Параллельная обработка callbacks из разных топиков
  // - Timer callback не блокирует обработку сообщений
  // - Лучшая отзывчивость при высокой нагрузке
  // 
  // ⚠️ ВАЖНО: Все shared state в контроллерах должны быть защищены mutex!
  rclcpp::executors::MultiThreadedExecutor executor;
  



//   // Вариант 1: Явное указание количества потоков
// rclcpp::executors::MultiThreadedExecutor executor(
//   rclcpp::ExecutorOptions(), 
//   4  // 4 потока
// );

// // Вариант 2: Использование количества CPU cores
// unsigned int num_threads = std::thread::hardware_concurrency();
// rclcpp::executors::MultiThreadedExecutor executor(
//   rclcpp::ExecutorOptions(), 
//   num_threads
// );

// // Вариант 3: По умолчанию (рекомендуется)
// // Executor сам выберет оптимальное количество
// rclcpp::executors::MultiThreadedExecutor executor;


  // === 5. Добавление узла в executor ===
  executor.add_node(los_node);
  
  // === 6. Запуск цикла обработки ===
  // spin() будет работать до тех пор, пока:
  // - Не будет вызван rclcpp::shutdown()
  // - Не будет нажат Ctrl+C
  // - Не произойдёт ошибка
  try {
    executor.spin();
  } catch (const std::exception &e) {
    RCLCPP_FATAL(los_node->get_logger(), "Executor exception: %s", e.what());
  }
  
  // === 7. Очистка ===
  RCLCPP_INFO(los_node->get_logger(), "LOS Guidance Node Shutting Down...");
  
  // === 8. Shutdown ROS2 ===
  rclcpp::shutdown();
  
  return 0;
}

void Node::controlLoopTimerCallback() {
  std::lock_guard<std::mutex> lock(state_mutex_);
  
  if (!attack_ready_) {
    return;
  }
  
  bool should_attack = (start_attack_ || attack_enabled_);
  
  if (should_attack) {
    if (vehicle_ctrl_->getCurrentMode() == FlightMode::OFFBOARD) {
      update_acc_setpoint_ = true;
      vehicle_ctrl_->publishOffboardControlMode();
      vehicle_ctrl_->publishAccelerationSetpoint(
        guidance_ctrl_->getLastAccelerationCommand()
      );
    } else {
      guidance_ctrl_->reset();
      vehicle_ctrl_->publishOffboardControlMode();
      vehicle_ctrl_->publishAccelerationSetpoint(
        guidance_ctrl_->getLastAccelerationCommand()
      );
      vehicle_ctrl_->setOffboardMode();
    }
  } else {
    if (vehicle_ctrl_->getCurrentMode() == FlightMode::OFFBOARD && update_acc_setpoint_) {
      guidance_ctrl_->reset();
      vehicle_ctrl_->setManualMode();
      update_acc_setpoint_ = false;
      RCLCPP_INFO(this->get_logger(), "Attack stopped, switched to MANUAL");
    }
  }
}

void SensorSync::registerSource(const std::string& name) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (buffers_.find(name) == buffers_.end()) {
    buffers_[name] = std::deque<std::pair<uint64_t, std::shared_ptr<void>>>();
    
    if (config_.debug) {
      std::cout << "[SensorSync] Registered source: " << name << std::endl;
    }
  }
}

void SensorSync::addMessage(
  const std::string& source,
  uint64_t ts_us,
  std::shared_ptr<void> msg
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  auto it = buffers_.find(source);
  if (it == buffers_.end()) {
    if (config_.debug) {
      std::cerr << "[SensorSync] Warning: Source '" << source 
                << "' not registered" << std::endl;
    }
    return;
  }
  
  auto& buf = it->second;
  buf.push_back({ts_us, msg});
  
  // Ограничение размера буфера
  while (buf.size() > config_.buffer_size) {
    buf.pop_front();
  }
  
  // Очистка старых сообщений
  while (!buf.empty() && buf.front().first < ts_us - config_.retention_us) {
    buf.pop_front();
  }
  
  if (config_.debug) {
    std::cout << "[SensorSync] Added message to '" << source 
              << "' at " << ts_us << " us, buffer size: " 
              << buf.size() << std::endl;
  }
}

std::map<std::string, std::shared_ptr<void>> SensorSync::sync(uint64_t ref_time) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  std::map<std::string, std::shared_ptr<void>> result;
  
  for (auto& [name, buf] : buffers_) {
    result[name] = getNearest_(name, ref_time);
    
    if (result[name]) {
      dropUntil_(name, ref_time);
    }
  }
  
  if (config_.debug) {
    std::cout << "[SensorSync] Sync completed at " << ref_time 
              << " us, sources: " << result.size() << std::endl;
  }
  
  return result;
}

void SensorSync::setDebug(bool enable) {
  std::lock_guard<std::mutex> lock(mutex_);
  config_.debug = enable;
}

// ============================================================================
// Приватные методы
// ============================================================================

std::shared_ptr<void> SensorSync::getNearest_(
  const std::string& source,
  uint64_t ref
) {
  auto it = buffers_.find(source);
  if (it == buffers_.end() || it->second.empty()) {
    return nullptr;
  }
  
  const auto& buf = it->second;
  std::shared_ptr<void> nearest = nullptr;
  uint64_t min_diff = std::numeric_limits<uint64_t>::max();
  uint64_t nearest_ts = 0;
  
  for (const auto& [ts, msg] : buf) {
    uint64_t diff = std::llabs(static_cast<int64_t>(ts - ref));
    if (diff < min_diff) {
      min_diff = diff;
      nearest = msg;
      nearest_ts = ts;
    }
  }
  
  if (config_.debug && nearest) {
    std::cout << "[SensorSync] Source: " << source
              << " | Ref: " << ref
              << " | Nearest: " << nearest_ts
              << " | Diff: " << min_diff << " us" << std::endl;
  }
  
  return nearest;
}

void SensorSync::dropUntil_(
  const std::string& source,
  uint64_t ref_time
) {
  auto it = buffers_.find(source);
  if (it == buffers_.end()) {
    return;
  }
  
  auto& buf = it->second;
  while (!buf.empty() && buf.front().first < ref_time) {
    buf.pop_front();
  }
}

TargetTracker::TargetTracker(const TrackerConfig& config)
  : config_(config)
{
  initCameraMatrices_();
}

void TargetTracker::initCameraMatrices_() {
  // Конвертация углов установки в радианы
  double roll_c = config_.install_roll_deg * M_PI / 180.0;
  double pitch_c = config_.install_pitch_deg * M_PI / 180.0;
  double yaw_c = config_.install_yaw_deg * M_PI / 180.0;
  
  // R_install (Body → rotated frame по углам установки)
  Eigen::Matrix3d R_bc_install =
    (Eigen::AngleAxisd(-yaw_c,   Eigen::Vector3d::UnitZ()) *
     Eigen::AngleAxisd(-pitch_c, Eigen::Vector3d::UnitY()) *
     Eigen::AngleAxisd(-roll_c,  Eigen::Vector3d::UnitX())).toRotationMatrix();
  
  // Axis swap: Body(FRD) → Camera(RDF)
  Eigen::Matrix3d axis_swap;
  axis_swap << 0, 1, 0,  // Body Y → Camera X
               0, 0, 1,  // Body Z → Camera Y
               1, 0, 0;  // Body X → Camera Z
  
  // Итоговая матрица body → camera
  R_bc_ = axis_swap * R_bc_install;
  R_cb_ = R_bc_.transpose();  // camera → body
}

TargetInfo TargetTracker::processTrackerResult(
  const ai_service::msg::TrackerResult::SharedPtr& msg,
  double roll_rad,
  double pitch_rad,
  const Eigen::Matrix3d& R_nb
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  TargetInfo info;
  info.timestamp_us = msg->timestamp / 1000;  // ns → us
  info.confidence = msg->target_info.confidence;
  
  // Проверка минимального доверия
  if (info.confidence < config_.min_confidence) {
    info.is_valid = false;
    stats_.frames_dropped++;
    return info;
  }
  
  // Пиксельные координаты (центр цели)
  info.pixel_u = (msg->target_info.left + msg->target_info.right) * config_.cx;
  info.pixel_v = (msg->target_info.top + msg->target_info.bottom) * config_.cy;
  info.bbox_width = std::abs(msg->target_info.right - msg->target_info.left);
  info.bbox_height = std::abs(msg->target_info.bottom - msg->target_info.top);
  
  // Сглаживание пикселей
  if (!pixel_initialized_) {
    info.pixel_u_smooth = info.pixel_u;
    info.pixel_v_smooth = info.pixel_v;
    pixel_initialized_ = true;
  } else {
    info.pixel_u_smooth = smooth_(pixel_u_prev_, info.pixel_u, config_.pixel_smooth_alpha);
    info.pixel_v_smooth = smooth_(pixel_v_prev_, info.pixel_v, config_.pixel_smooth_alpha);
  }
  pixel_u_prev_ = info.pixel_u_smooth;
  pixel_v_prev_ = info.pixel_v_smooth;
  
  // Коррекция на roll/pitch дрона
  auto [u_correct, v_correct] = correctPixelRollPitch_(
    info.pixel_u_smooth, info.pixel_v_smooth,
    roll_rad, pitch_rad
  );
  
  // Pixel → Camera
  Eigen::Vector3d u_c = pixelToCamVec_(u_correct, v_correct);
  info.los_cam = u_c;
  
  // Camera → Nav
  info.los_nav = camToNav_(u_c,  R_nb);//R_bc_,
  
  // Сглаживание LOS вектора
  if (!los_initialized_) {
    los_nav_prev_ = info.los_nav;
    los_initialized_ = true;
  }
  info.los_nav = smooth_(los_nav_prev_, info.los_nav, config_.los_smooth_alpha);
  los_nav_prev_ = info.los_nav;
  
  // Расчёт угловой скорости LOS (lambda)
  double dt = (last_timestamp_us_ > 0) ? 
    (info.timestamp_us - last_timestamp_us_) * 1e-6 : 0.01;
  info.los_lambda = computeLosLambda_(info.los_nav, los_nav_history_, dt);
  los_nav_history_ = info.los_nav;
  last_timestamp_us_ = info.timestamp_us;
  
  info.is_valid = true;
  stats_.frames_processed++;
  
  return info;
}

TargetInfo TargetTracker::processDetection2D(
  const vision_msgs::msg::Detection2D::SharedPtr& msg,
  double roll_rad,
  double pitch_rad,
  const Eigen::Matrix3d& R_nb
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  TargetInfo info;
  info.timestamp_us = msg->header.stamp.sec * 1e6 + msg->header.stamp.nanosec * 1e-3;
  info.confidence = 1.0;  // Detection2D не имеет confidence
  
  // Пиксельные координаты из bbox
  info.pixel_u = msg->bbox.center.position.x;
  info.pixel_v = msg->bbox.center.position.y;
  info.bbox_width = msg->bbox.size_x;
  info.bbox_height = msg->bbox.size_y;
  
  // Сглаживание пикселей
  if (!pixel_initialized_) {
    info.pixel_u_smooth = info.pixel_u;
    info.pixel_v_smooth = info.pixel_v;
    pixel_initialized_ = true;
  } else {
    info.pixel_u_smooth = smooth_(pixel_u_prev_, info.pixel_u, config_.pixel_smooth_alpha);
    info.pixel_v_smooth = smooth_(pixel_v_prev_, info.pixel_v, config_.pixel_smooth_alpha);
  }
  pixel_u_prev_ = info.pixel_u_smooth;
  pixel_v_prev_ = info.pixel_v_smooth;
  
  // Коррекция на roll/pitch
  auto [u_correct, v_correct] = correctPixelRollPitch_(
    info.pixel_u_smooth, info.pixel_v_smooth,
    roll_rad, pitch_rad
  );
  
  // Pixel → Camera → Nav
  Eigen::Vector3d u_c = pixelToCamVec_(u_correct, v_correct);
  info.los_cam = u_c;
  info.los_nav = camToNav_(u_c,  R_nb); //R_bc_,
  
  // Сглаживание LOS
  if (!los_initialized_) {
    los_nav_prev_ = info.los_nav;
    los_initialized_ = true;
  }
  info.los_nav = smooth_(los_nav_prev_, info.los_nav, config_.los_smooth_alpha);
  los_nav_prev_ = info.los_nav;
  
  // Lambda
  double dt = (last_timestamp_us_ > 0) ?
    (info.timestamp_us - last_timestamp_us_) * 1e-6 : 0.01;
  info.los_lambda = computeLosLambda_(info.los_nav, los_nav_history_, dt);
  los_nav_history_ = info.los_nav;
  last_timestamp_us_ = info.timestamp_us;
  
  info.is_valid = true;
  stats_.frames_processed++;
  
  return info;
}

Eigen::Vector3d TargetTracker::pixelToCamVec_(double u, double v) {
  Eigen::Vector3d vec;
  vec << (u - config_.cx) / config_.fx,
         (v - config_.cy) / config_.fy,
         1.0;
  return vec.normalized();
}

std::pair<double, double> TargetTracker::correctPixelRollPitch_(
  double u, double v,
  double roll_rad, double pitch_rad
) {
  // Back-project pixel to camera ray
  double x = (u - config_.cx) / config_.fx;
  double y = (v - config_.cy) / config_.fy;
  Eigen::Vector3d r_cam(x, y, 1.0);
  r_cam.normalize();
  
  // Rotation matrix для коррекции (undo roll/pitch)
  Eigen::Matrix3d R_body_corr =
    (Eigen::AngleAxisd(pitch_rad, Eigen::Vector3d::UnitY()) *
     Eigen::AngleAxisd(roll_rad,  Eigen::Vector3d::UnitX())).toRotationMatrix();
  
  // Apply correction: R_bc * R_corr * R_bc^T * r_cam
  Eigen::Vector3d r_corr = R_bc_ * R_body_corr * R_cb_ * r_cam;
  
  // Protection against degenerate case (z ~ 0)
  const double EPS_Z = 1e-4;
  if (std::abs(r_corr.z()) < EPS_Z) {
    return {u, v};  // Fallback: no correction
  }
  
  // Project back to pixel coordinates
  double u_correct = config_.cx + config_.fx * (r_corr.x() / r_corr.z());
  double v_correct = config_.cy + config_.fy * (r_corr.y() / r_corr.z());
  
  // Clamp to image plane
  u_correct = std::clamp(u_correct, 0.0, 2.0 * config_.cx);
  v_correct = std::clamp(v_correct, 0.0, 2.0 * config_.cy);
  
  return {u_correct, v_correct};
}

Eigen::Vector3d TargetTracker::camToNav_(
  const Eigen::Vector3d& u_c,
  // const Eigen::Matrix3d& R_bc,
  const Eigen::Matrix3d& R_nb
) {
  Eigen::Vector3d u_b = R_cb_ * u_c;      // camera → body
  Eigen::Vector3d u_n = R_nb.transpose() * u_b;  // body → nav
  return u_n.normalized();
}

Eigen::Vector3d TargetTracker::computeLosLambda_(
  const Eigen::Vector3d& los_current,
  const Eigen::Vector3d& los_prev,
  double dt
) {
  // lambda = u_n × u_dot_n (cross product)
  Eigen::Vector3d u_dot = (los_current - los_prev) / dt;
  Eigen::Vector3d lambda = los_current.cross(u_dot);
  return lambda;
}

double TargetTracker::smooth_(double prev, double current, double alpha) {
  return prev + alpha * (current - prev);
}

Eigen::Vector3d TargetTracker::smooth_(
  const Eigen::Vector3d& prev,
  const Eigen::Vector3d& current,
  double alpha
) {
  return prev + alpha * (current - prev);
}

Eigen::Matrix3d TargetTracker::skewSymmetric_(const Eigen::Vector3d& v) {
  Eigen::Matrix3d m;
  m <<     0, -v.z(),  v.y(),
         v.z(),     0, -v.x(),
        -v.y(),  v.x(),     0;
  return m;
}

VehicleController::VehicleController(
  rclcpp::Node::SharedPtr node,
  const VehicleControllerConfig& config
)
  : node_(node)
  , config_(config)
{
  if (!node_) {
    throw std::runtime_error("VehicleController: node pointer is null");
  }
  
  // Создание publishers
  createPublishers_();
  
  // Инициализация состояния
  reset();
  
  if (config_.debug_logging) {
    RCLCPP_INFO(node_->get_logger(), "VehicleController initialized");
  }
}

void VehicleController::updateStatus(
  const px4_msgs::msg::VehicleStatus::SharedPtr msg
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (!msg) {
    return;
  }
  
  status_.nav_state = msg->nav_state;
  status_.flight_mode = navStateToFlightMode_(msg->nav_state);
  status_.is_armed = (msg->arming_state == px4_msgs::msg::VehicleStatus::ARMING_STATE_ARMED);
  status_.last_status_timestamp_us = msg->timestamp;
  status_.is_valid = true;
  
  // Offboard доступен только в режиме OFFBOARD
  status_.is_offboard_available = 
    (status_.flight_mode == FlightMode::OFFBOARD) && status_.is_armed;
  
  if (config_.debug_logging) {
    RCLCPP_DEBUG(
      node_->get_logger(),
      "Vehicle status: mode=%s, armed=%s, offboard=%s",
      flightModeToString(status_.flight_mode).c_str(),
      status_.is_armed ? "true" : "false",
      status_.is_offboard_available ? "true" : "false"
    );
  }
}

bool VehicleController::setOffboardMode() {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (status_.flight_mode == FlightMode::OFFBOARD) {
    if (config_.debug_logging) {
      RCLCPP_DEBUG(node_->get_logger(), "Already in OFFBOARD mode");
    }
    return true;
  }
  
  bool success = publishVehicleCommand_(
    VEHICLE_CMD_DO_SET_MODE,
    1.0,  // Base mode
    static_cast<float>(PX4_CUSTOM_MAIN_MODE_OFFBOARD)
  );
  
  if (success) {
    offboard_mode_sent_ = true;
    last_command_timestamp_us_ = getCurrentTimestamp_();
    
    if (config_.debug_logging) {
      RCLCPP_INFO(node_->get_logger(), "OFFBOARD mode command sent");
    }
  }
  
  return success;
}

bool VehicleController::publishAccelerationSetpoint(
  const Eigen::Vector3d& acc_setpoint
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (!status_.is_valid) {
    RCLCPP_WARN(node_->get_logger(), "Cannot publish: status not valid");
    return false;
  }
  
  // Сглаживание ускорения
  Eigen::Vector3d acc_smooth = smoothAcceleration_(acc_setpoint);
  
  // Ограничение ускорения
  acc_smooth = clampAcceleration_(acc_smooth);
  
  // Публикация offboard control mode (должна публиковаться регулярно)
  publishOffboardControlMode();
  
  // Публикация trajectory setpoint
  px4_msgs::msg::TrajectorySetpoint msg;
  msg.position = {NAN, NAN, NAN};
  msg.velocity = {NAN, NAN, NAN};
  msg.acceleration = {
    static_cast<float>(acc_smooth.x()),
    static_cast<float>(acc_smooth.y()),
    static_cast<float>(acc_smooth.z())
  };
  msg.yaw = NAN;
  msg.timestamp = getCurrentTimestamp_();
  
  traj_pub_->publish(msg);
  
  // Сохранение последней уставки
  last_setpoint_.acceleration = acc_smooth;
  last_setpoint_.timestamp_us = msg.timestamp;
  last_setpoint_.is_valid = true;
  
  if (config_.debug_logging) {
    RCLCPP_DEBUG(
      node_->get_logger(),
      "Acc setpoint published: [%.3f, %.3f, %.3f]",
      acc_smooth.x(), acc_smooth.y(), acc_smooth.z()
    );
  }
  
  return true;
}

  msg.position = config_.enable_position_control;
  msg.velocity = config_.enable_velocity_control;

bool VehicleController::publishTrajectorySetpoint(
  const ControlSetpoint& setpoint
) {
  std::lock_guard<std::mutex> lock(mutex_);
  
  if (!setpoint.is_valid) {
    return false;
  }
  
  publishOffboardControlMode();
  
  px4_msgs::msg::TrajectorySetpoint msg;
  msg.position = {
    static_cast<float>(setpoint.position.x()),
    static_cast<float>(setpoint.position.y()),
    static_cast<float>(setpoint.position.z())
  };
  msg.velocity = {
    static_cast<float>(setpoint.velocity.x()),
    static_cast<float>(setpoint.velocity.y()),
    static_cast<float>(setpoint.velocity.z())
  };
  msg.acceleration = {
    static_cast<float>(setpoint.acceleration.x()),
    static_cast<float>(setpoint.acceleration.y()),
    static_cast<float>(setpoint.acceleration.z())
  };
  msg.yaw = std::isnan(setpoint.yaw) ? NAN : static_cast<float>(setpoint.yaw);
  msg.timestamp = getCurrentTimestamp_();
  
  traj_pub_->publish(msg);
  
  last_setpoint_ = setpoint;
  
  return true;
}

std::string VehicleController::flightModeToString(FlightMode mode) {
  switch (mode) {
    case FlightMode::UNKNOWN: return "UNKNOWN";
    case FlightMode::OFFBOARD: return "OFFBOARD";
    case FlightMode::MANUAL: return "MANUAL";
    case FlightMode::ALTCTL: return "ALTCTL";
    case FlightMode::POSCTL: return "POSCTL";
    case FlightMode::AUTO: return "AUTO";
    case FlightMode::ACRO: return "ACRO";
    case FlightMode::STABILIZED: return "STABILIZED";

  }
}

void VehicleController::createPublishers_() {
  cmd_pub_ = node_->create_publisher<px4_msgs::msg::VehicleCommand>(
    "fmu/in/vehicle_command", 10);
  
  offboard_pub_ = node_->create_publisher<px4_msgs::msg::OffboardControlMode>(
    "fmu/in/offboard_control_mode", 10);
  
  traj_pub_ = node_->create_publisher<px4_msgs::msg::TrajectorySetpoint>(
    "fmu/in/trajectory_setpoint", 10);
  
  if (config_.debug_logging) {
    RCLCPP_INFO(node_->get_logger(), "VehicleController publishers created");
  }
}

  msg.param3 = 0.0;
  msg.param4 = 0.0;
  msg.param5 = 0.0;
  msg.param6 = 0.0;
  msg.param7 = 0.0;
  msg.command = command;
  msg.target_system = config_.target_system;
  msg.target_component = config_.target_component;
  msg.source_system = config_.source_system;
  msg.source_component = config_.source_component;
  msg.from_external = true;
  msg.timestamp = getCurrentTimestamp_();
  
  cmd_pub_->publish(msg);
  
  return true;
}

Eigen::Vector3d VehicleController::smoothAcceleration_(
  const Eigen::Vector3d& acc_raw
) {
  if (!acc_initialized_) {
    acc_smoothed_ = acc_raw;
    acc_initialized_ = true;
    return acc_raw;
  }
  
  // 🆕 Адаптивное сглаживание для тяжёлого дрона (2519г)
  double acc_change = (acc_raw - acc_smoothed_).norm();
  double adaptive_alpha = config_.acc_smooth_alpha;
  
  // Более консервативно для тяжёлого
  if (acc_change > 1.0) {  // ↓ Меньше порог
    adaptive_alpha = 0.15;  // ↓ Меньше сглаживание при резких изменениях
  } else if (acc_change < 0.2) {
    adaptive_alpha = 0.04;  // ↓ Больше сглаживание при плавных
  }
  
  acc_smoothed_ = acc_smoothed_ + adaptive_alpha * (acc_raw - acc_smoothed_);
  return acc_smoothed_;
}



Eigen::Vector3d VehicleController::clampAcceleration_(
  const Eigen::Vector3d& acc
) {
  Eigen::Vector3d acc_clamped;
  
  // 🆕 Для тяжёлого дрона (2519г) - более консервативные лимиты
  double heavy_drone_factor = 0.62;  // = 1562/2519
  
  acc_clamped.x() = std::clamp(
    acc.x(),
    -config_.max_hor_acc_limit * heavy_drone_factor,
    config_.max_hor_acc_limit * heavy_drone_factor
  );
  acc_clamped.y() = std::clamp(
    acc.y(),
    -config_.max_hor_acc_limit * heavy_drone_factor,
    config_.max_hor_acc_limit * heavy_drone_factor
  );
  acc_clamped.z() = std::clamp(
    acc.z(),
    config_.max_up_acc * heavy_drone_factor,
    config_.max_down_acc * heavy_drone_factor
  );
  
  return acc_clamped;
}


uint64_t VehicleController::getCurrentTimestamp_() const {
  auto now = node_->get_clock()->now();
  return static_cast<uint64_t>(now.nanoseconds() / 1000);  // ns → us
}

FlightMode VehicleController::navStateToFlightMode_(uint8_t nav_state) const {
  using NavState = px4_msgs::msg::VehicleStatus;
  
  switch (nav_state) {
    case NavState::NAVIGATION_STATE_OFFBOARD:
      return FlightMode::OFFBOARD;
    
    case NavState::NAVIGATION_STATE_MANUAL:
    case NavState::NAVIGATION_STATE_STAB:
      return FlightMode::MANUAL;
    
    case NavState::NAVIGATION_STATE_ALTCTL:
      return FlightMode::ALTCTL;
    
    case NavState::NAVIGATION_STATE_POSCTL:
      return FlightMode::POSCTL;
    
    case NavState::NAVIGATION_STATE_AUTO_MISSION:
    case NavState::NAVIGATION_STATE_AUTO_LOITER:
    case NavState::NAVIGATION_STATE_AUTO_RTL:
    case NavState::NAVIGATION_STATE_AUTO_TAKEOFF:
    case NavState::NAVIGATION_STATE_AUTO_LAND:
    case NavState::NAVIGATION_STATE_AUTO_FOLLOW_TARGET:
    case NavState::NAVIGATION_STATE_AUTO_PRECLAND:
      return FlightMode::AUTO;
    
    case NavState::NAVIGATION_STATE_ACRO:
      return FlightMode::ACRO;
    

    
    default:
      return FlightMode::OTHERS;
  }
}
